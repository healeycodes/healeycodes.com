ground (Accelerate)
wishspd = wishspeed           // no cap: 250
// speed already going that way:
already = dot(v, wishdir)
// how much more it will give:
room    = wishspd - already
if room <= 0: return
step    = accel    * wishspeed * dt
v       += min(step, room) * wishdir
air (AirAccelerate)
wishspd = min(wishspeed, 30)  // cap: 30
// speed already going that way:
already = dot(v, wishdir)
// how much more it will give:
room    = wishspd - already
if room <= 0: return
step    = airaccel * wishspeed * dt
v       += min(step, room) * wishdir
